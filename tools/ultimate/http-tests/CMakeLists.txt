cmake_minimum_required(VERSION 3.20)
project(ultimate_http_timeouts LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
set(REPO_ROOT ${CMAKE_CURRENT_SOURCE_DIR}/../../..)
add_executable(http_timeout_policy test_timeouts.cpp)
target_include_directories(http_timeout_policy PRIVATE ${REPO_ROOT}/emulator/src)
if(MSVC)
  target_compile_options(http_timeout_policy PRIVATE /W4 /WX)
else()
  target_compile_options(http_timeout_policy PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME timeout_policy COMMAND http_timeout_policy)
option(ULTIMATE_LOCAL_TLS_TEST "Run real httplib/OpenSSL against a delayed local TLS server" OFF)
if(ULTIMATE_LOCAL_TLS_TEST)
  find_package(Python3 REQUIRED COMPONENTS Interpreter)
  find_package(OpenSSL 3 REQUIRED)
  find_package(Threads REQUIRED)
  find_program(OPENSSL_EXECUTABLE openssl REQUIRED)
  add_executable(http_tls_client tls_client.cpp)
  target_include_directories(http_tls_client PRIVATE ${REPO_ROOT}/emulator/src)
  target_include_directories(http_tls_client SYSTEM PRIVATE ${REPO_ROOT}/bbport/gpu/third_party/httplib)
  target_compile_definitions(http_tls_client PRIVATE CPPHTTPLIB_OPENSSL_SUPPORT)
  target_link_libraries(http_tls_client PRIVATE OpenSSL::SSL OpenSSL::Crypto Threads::Threads)
  add_test(NAME delayed_verified_tls
    COMMAND ${Python3_EXECUTABLE} ${CMAKE_CURRENT_SOURCE_DIR}/delayed_tls.py
            $<TARGET_FILE:http_tls_client> ${OPENSSL_EXECUTABLE})
  set_tests_properties(delayed_verified_tls PROPERTIES TIMEOUT 40)
endif()
