cmake_minimum_required(VERSION 3.20)
project(UltimateGameLinkTests LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
enable_testing()
get_filename_component(REPO_ROOT "${CMAKE_CURRENT_LIST_DIR}/../../.." ABSOLUTE)
set(ULTIMATE_GTEST_SOURCE "" CACHE PATH "Local checkout of GoogleTest")
if(NOT EXISTS "${ULTIMATE_GTEST_SOURCE}/CMakeLists.txt")
    message(FATAL_ERROR "Set ULTIMATE_GTEST_SOURCE to a GoogleTest checkout; no downloads occur here")
endif()
set(gtest_force_shared_crt ON CACHE BOOL "" FORCE)
set(INSTALL_GTEST OFF CACHE BOOL "" FORCE)
add_subdirectory("${ULTIMATE_GTEST_SOURCE}" "${CMAKE_BINARY_DIR}/googletest" EXCLUDE_FROM_ALL)
add_executable(game_link_existing "${REPO_ROOT}/emulator/tests/bloodborne/test_bloodborne_game_link.cpp")
target_include_directories(game_link_existing PRIVATE "${REPO_ROOT}/server" "${REPO_ROOT}/emulator/src")
target_link_libraries(game_link_existing PRIVATE GTest::gtest_main)
add_test(NAME game_link_existing COMMAND game_link_existing)
add_executable(game_link_restart "${REPO_ROOT}/server/tests/test_game_link_restart.cpp")
target_include_directories(game_link_restart PRIVATE "${REPO_ROOT}/server")
add_test(NAME game_link_restart COMMAND game_link_restart)
set_tests_properties(game_link_existing game_link_restart PROPERTIES TIMEOUT 30)
add_executable(game_link_resend "${REPO_ROOT}/server/tests/test_game_link_resend.cpp")
target_include_directories(game_link_resend PRIVATE "${REPO_ROOT}/server")
add_test(NAME game_link_resend COMMAND game_link_resend)
set_tests_properties(game_link_resend PROPERTIES TIMEOUT 30)
add_executable(game_link_allocations "${REPO_ROOT}/server/tests/test_game_link_allocations.cpp")
target_include_directories(game_link_allocations PRIVATE "${REPO_ROOT}/server")
add_test(NAME game_link_allocations COMMAND game_link_allocations)
set_tests_properties(game_link_allocations PROPERTIES TIMEOUT 30)
add_executable(game_link_transcript "${CMAKE_CURRENT_LIST_DIR}/wire_transcript.cpp")
target_include_directories(game_link_transcript PRIVATE "${REPO_ROOT}/server")
add_test(NAME game_link_transcript COMMAND game_link_transcript)
set_tests_properties(game_link_transcript PROPERTIES TIMEOUT 30)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
execute_process(COMMAND "${Python3_EXECUTABLE}"
    "${REPO_ROOT}/tools/re/tests/test_team_world_preservation.py"
    --emit "${CMAKE_BINARY_DIR}/team_world_fixture.cpp"
    COMMAND_ERROR_IS_FATAL ANY)
add_executable(team_world_preservation "${CMAKE_BINARY_DIR}/team_world_fixture.cpp")
add_test(NAME team_world_preservation COMMAND team_world_preservation)
set_tests_properties(team_world_preservation PROPERTIES TIMEOUT 30)
