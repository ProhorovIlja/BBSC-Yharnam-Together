cmake_minimum_required(VERSION 3.20)
project(ultimate_dlss_camera LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(dlss_camera test_camera.cpp)
target_include_directories(dlss_camera PRIVATE ../../../bbport/gpu/dlss_bridge)
if(MSVC)
  target_compile_options(dlss_camera PRIVATE /W4 /WX /utf-8)
else()
  target_compile_options(dlss_camera PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME dlss_camera_projections COMMAND dlss_camera)
