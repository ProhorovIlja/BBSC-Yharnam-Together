cmake_minimum_required(VERSION 3.20)
project(UltimateLodGuard LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
set(CMAKE_CXX_EXTENSIONS OFF)
find_package(Python3 REQUIRED COMPONENTS Interpreter)
find_package(Threads REQUIRED)
execute_process(COMMAND "${Python3_EXECUTABLE}" "${CMAKE_CURRENT_SOURCE_DIR}/generate_fixture.py"
    "${CMAKE_BINARY_DIR}/lod_bytes.h" COMMAND_ERROR_IS_FATAL ANY)
add_executable(lod_claim "${CMAKE_CURRENT_SOURCE_DIR}/../../../bbport/tests/lod_claim_harness.cpp")
target_include_directories(lod_claim PRIVATE "${CMAKE_BINARY_DIR}")
target_link_libraries(lod_claim PRIVATE Threads::Threads)
enable_testing()
add_test(NAME actual_lod_claim COMMAND lod_claim patched 20000)
set_tests_properties(actual_lod_claim PROPERTIES TIMEOUT 120 SKIP_RETURN_CODE 77)
