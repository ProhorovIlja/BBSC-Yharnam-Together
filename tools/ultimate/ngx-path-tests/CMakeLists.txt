cmake_minimum_required(VERSION 3.20)
project(ultimate_ngx_paths LANGUAGES CXX)
set(CMAKE_CXX_STANDARD 20)
set(CMAKE_CXX_STANDARD_REQUIRED ON)
enable_testing()
add_executable(ngx_paths test_paths.cpp)
target_include_directories(ngx_paths PRIVATE ../../../bbport/gpu/shim)
if(MSVC)
  target_compile_options(ngx_paths PRIVATE /W4 /WX /utf-8)
  target_compile_definitions(ngx_paths PRIVATE _CRT_SECURE_NO_WARNINGS)
else()
  target_compile_options(ngx_paths PRIVATE -Wall -Wextra -Werror)
endif()
add_test(NAME locale_independent_ngx_paths COMMAND ngx_paths)
